// Version: v1.7.10 - Service Worker const CACHE_NAME = 'shir-shel-yom-v1.7.10'; const urlsToCache = [ './', './index.html', './manifest.json' ]; // התקנה וניקון מטמון ישן self.addEventListener('install', event => { self.skipWaiting(); // דחיפה מיידית של ה-SW החדש event.waitUntil( caches.open(CACHE_NAME) .then(cache => cache.addAll(urlsToCache)) ); }); // הפעלה ומחיקת קכשים ישנים self.addEventListener('activate', event => { event.waitUntil( caches.keys().keys().then(cacheNames => { return Promise.all( cacheNames.map(cacheName => { if (cacheName !== CACHE_NAME) { return caches.delete(cacheName); // מוחק כל מטמון ישן } }) ); }).then(() => self.clients.claim()) ); }); // שליפת נתונים ברשת / אופליין self.addEventListener('fetch', event => { event.respondWith( caches.match(event.request) .then(response => response || fetch(event.request)) ); });
